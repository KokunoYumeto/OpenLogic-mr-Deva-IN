\olsection{\usetoken{S}{derivability} आणि संख्यापक}

\begin{explain}
  संपूर्णता प्रमेयासाठी या विभागात स्थापित केलेली~$\Proves$ संबंधी तथ्ये
  टॅब्लोच्या नियमांनी निष्पन्न होतात, हेही आवश्यक आहे.
\end{explain}

\begin{thm}
\ollabel{thm:strong-generalization} जर $c$ हे $\Gamma$ किंवा~$!A(x)$ मध्ये
न येणारे अचर असेल आणि $\Gamma \Proves !A(c)$, तर $\Gamma
\Proves \lforall[x][!A(x)]$.
\end{thm}

\begin{proof}
$\Gamma \Proves !A(c)$ असे समजा; म्हणजे $!B_1$, \dots, $!B_n
\in \Gamma$ अशी वाक्ये आणि पुढील संचासाठी बंद !!{tableau} असतो:
\begin{align*}
\{ \sFmla{\False}{!A(c)}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\intertext{पुढील संचासाठीही बंद !!{tableau} असतो, हे दाखवायचे आहे:}
\{ \sFmla{\False}{\lforall[x][!A(x)]}, &
\sFmla{\True}{!B_1}, \dots, \sFmla{\True}{!B_n} \}.
\end{align*}
बंद !!{tableau} घेऊन त्यातील पहिले गृहीतक
$\sFmla{\False}{\lforall[x][!A(x)]}$ ने बदला आणि गृहीतकांनंतर
$\sFmla{\False}{!A(c)}$ घाला.
\begin{center}
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\formula{A}(c)}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
      [\sFmla{\True}{\formula{B}_n}, 
        [][][]]]]]
\end{tableau}\qquad
\begin{tableau}{not line numbering}
  [\sFmla{\False}{\lforall[x][\formula{A}(x)]}
    [\sFmla{\True}{\formula{B}_1}
      [\vdots
        [\sFmla{\True}{\formula{B}_n},
          [\sFmla{\False}{\formula{A}(c)}
        [][][]]]]]]
\end{tableau}
\end{center}
हा टॅब्लो अजूनही बंद आहे, कारण पूर्वी गृहीतके म्हणून उपलब्ध असलेली सर्व
!!{sentence}s नव्या !!{tableau} च्या वरच्या भागात अजूनही उपलब्ध आहेत. घातलेली
ओळ ही $\TRule{\False}{\lforall}$ च्या योग्य उपयोगाचा परिणाम आहे, कारण
!!{constant}~$c$ हे $!B_1$, \dots, $!B_n$ किंवा
$\lforall[x][!A(x)]$ मध्ये येत नाही; म्हणजे ते नव्या !!{tableau} मध्ये घातलेल्या
ओळीच्या वर येत नाही.
\end{proof}

\begin{prop}
\ollabel{prop:provability-quantifiers}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{$!A(t) \Proves \lexists[x][!A(x)]$.}{}
\tagitem{prvAll}{$\lforall[x][!A(x)] \Proves !A(t)$.}{}
\end{tagenumerate}
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvEx,prvAll}
\tagitem{prvEx}{
  $\sFmla{\False}{\lexists[x][!A(x)]}, \sFmla{\True}{!A(t)}$ साठीचा बंद
  !!{tableau} पुढीलप्रमाणे आहे:
  \begin{oltableau}
    [\sFmla{\False}{\lexists[x][\formula{A}(x)]}, just = \TAss
      [\sFmla{\True}{\formula{A}(t)}, just = \TAss
        [\sFmla{\False}{\formula{A}(t)}, just = {\TRule{\False}{\lexists}[1]}, close]]]
    \end{oltableau}}{}
\tagitem{prvAll}{
  $\sFmla{\False}{\formula{A}(t)}, \sFmla{\True}{\lforall[x][!A(x)]}, $
  साठीचा बंद !!{tableau} पुढीलप्रमाणे आहे:
  \begin{oltableau}
    [\sFmla{\False}{\formula{A}(t)}, just = \TAss
      [\sFmla{\True}{\lforall[x][\formula{A}(x)]}, just = \TAss
        [\sFmla{\True}{\formula{A}(t)}, just = {\TRule{\True}{\lforall}[2]}, close]]]
    \end{oltableau}}{}
\end{tagenumerate}
\end{proof}
