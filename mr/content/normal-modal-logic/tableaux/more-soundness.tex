\olsection{अतिरिक्त नियमांची निर्दोषता}

प्रतिमानांच्या एखाद्या वर्गातील एका प्रतिमानात
!!a{tableau} ची शाखा पूर्ततायोग्य असेल, तेव्हा
नियम लावल्यावर मिळणारी शाखाही त्या वर्गातील
प्रतिमानात पूर्ततायोग्य असेल, तर तो नियम त्या
वर्गासाठी निर्दोष आहे असे म्हणतो.

\begin{prop}
  \ollabel{prop:soundness-T}
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ आणि }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  परावर्ती प्रतिमानांसाठी
  \iftag{notprvBox,notprvDiamond}{निर्दोष आहे}{निर्दोष आहेत}.
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{$\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$
  याला T$\Box$ लावून शाखा विस्तारित केली:
  \iftag{probBox}{सरावासाठी.}{शाखेवर
    $\sFmla{\True}{!B}[\sigma]$ हे नवे !!{signed
      formula} मिळते. $\mSat{M}{\Gamma}[f]$
    असे समजू; विशेषतः $\mSat{M}{\Box
      !B}[f(\sigma)]$. $R$ परावर्ती असल्याने
    $Rf(\sigma)f(\sigma)$. म्हणून
    $\mSat{M}{!B}[f(\sigma)]$; म्हणजे
    $\mModel{M}, f$ हे
    $\sFmla{\True}{!B}[\sigma]$ पूर्ण करते.}}{}
\tagitem{prvDiamond}{$\sFmla{\False}{\Diamond !B}[\sigma] \in \Gamma$
  याला T$\Diamond$ लावून शाखा विस्तारित केली:
  \iftag{probDiamond}{सरावासाठी.}{शाखेवर
    $\sFmla{\False}{!B}[\sigma]$ हे नवे !!{signed
      formula} मिळते. $\mSat{M}{\Gamma}[f]$
    असे समजू; विशेषतः $\mSat/{M}{\Diamond
      !B}[f(\sigma)]$. $R$ परावर्ती असल्याने
    $Rf(\sigma)f(\sigma)$. म्हणून
    $\mSat/{M}{!B}[f(\sigma)]$; म्हणजे
    $\mModel{M}, f$ हे
    $\sFmla{\False}{!B}[\sigma]$ पूर्ण करते.}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  \olref[nml][tab][msn]{prop:soundness-T} चा पुरावा पूर्ण करा.
\end{prob}
\tagendprob
