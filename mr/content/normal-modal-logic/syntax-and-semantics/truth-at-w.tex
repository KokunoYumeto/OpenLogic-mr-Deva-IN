\olsection{एखाद्या जगात सत्य असणे}

प्रत्येक मोडल प्रतिमान त्यातील कोणती मोडल !!{formula}s कोणत्या
जगांत सत्य मानायची हे ठरवते. ``प्रतिमान~$\mModel{M}$ मध्ये
!!{formula}~$!A$ जग~$w$ येथे सत्य आहे'' हा संबंधी
चिन्हार्थमीमांसेतील मूलभूत संबंध आहे. हा संबंध विगमनाने
परिभाषित केला जातो; मोडल नसलेल्या परिकर्मींसाठी तो
सत्यताकोष्टकांनी दिलेल्या नेहमीच्या लक्षणाशी जुळतो.

\begin{defn}\ollabel{defn:mmodels}
  प्रतिमान~$\mModel M$ मध्ये \emph{!!a{formula}~$!A$ हे~$w$ येथे
  सत्य असणे}, म्हणजे $\mSat{M}{!A}[w]$, पुढीलप्रमाणे
  विगमनाने परिभाषित करतो:
  \begin{enumerate}
  \tagitem{prvFalse}{\indcase{!A}{\lfalse}{$\mSat{M}{\lfalse}[w]$ कधीच नसते}.}{}
  \tagitem{prvTrue}{\indcase{!A}{\ltrue}{$\mSat{M}{\ltrue}[w]$ नेहमी असते}.}{}
  \item $\mSat{M}{p}[w]$ तेव्हा आणि केवळ तेव्हाच, जेव्हा $w \in V(p)$.
  \tagitem{prvNot}{\indcase{!A}{\lnot !B}{$\mSat{M}{\indfrm}[w]$
    तेव्हा आणि केवळ तेव्हाच, जेव्हा $\mSat/{M}{!B}[w]$}.}{}
  \tagitem{prvAnd}{\indcase{!A}{(!B \land !C)}{$\mSat{M}{\indfrm}[w]$
    तेव्हा आणि केवळ तेव्हाच, जेव्हा $\mSat{M}{!B}[w]$ आणि
    $\mSat{M}{!C}[w]$}.}{}
  \tagitem{prvOr}{\indcase{!A}{(!B \lor !C)}{$\mSat{M}{\indfrm}[w]$
    तेव्हा आणि केवळ तेव्हाच, जेव्हा $\mSat{M}{!B}[w]$ किंवा
    $\mSat{M}{!C}[w]$} (किंवा दोन्ही).}{}
  \tagitem{prvIf}{\indcase{!A}{(!B \lif !C)}{$\mSat{M}{\indfrm}[w]$
    तेव्हा आणि केवळ तेव्हाच, जेव्हा $\mSat/{M}{!B}[w]$ किंवा
    $\mSat{M}{!C}[w]$}.}{}
  \tagitem{prvIff}{\indcase{!A}{(!B \liff !C)}{$\mSat{M}{\indfrm}[w]$
    तेव्हा आणि केवळ तेव्हाच, जेव्हा $\mSat{M}{!B}[w]$ आणि
    $\mSat{M}{!C}[w]$ ही दोन्ही असतात, किंवा
    $\mSat{M}{!B}[w]$ आणि $\mSat{M}{!C}[w]$ यांपैकी कोणतेही नसते}.}{}
  \item\ollabel{defn:sub:mmodels-box}
    \indcase{!A}{\Box !B}{$\mSat{M}{\indfrm}[w]$ तेव्हा आणि
    केवळ तेव्हाच, जेव्हा $Rww'$ असलेल्या प्रत्येक $w' \in W$
    साठी $\mSat{M}{!B}[w']$.}
  \item\ollabel{defn:sub:mmodels-diamond}
    \indcase{!A}{\Diamond !B}{$\mSat{M}{\indfrm}[w]$ तेव्हा आणि
    केवळ तेव्हाच, जेव्हा $Rww'$ असलेल्या किमान एका $w' \in W$
    साठी $\mSat{M}{!B}[w']$.}
  \end{enumerate}
\end{defn}

\olref{defn:sub:mmodels-box} या खंडानुसार, $Rww'$ असलेले
एकही~$w'$ नसेल तेव्हा !!a{formula}~$\Box !B$ हे~$w$ येथे
सत्य असते, हे लक्षात घ्या. अशा वेळी $\Box !B$ हे~$w$ येथे
\emph{रिक्तपणे} सत्य असते. शिवाय $!B$ सत्य नसतानाही
$\Box !B$ ची~$w$ येथे पूर्ति होऊ शकते. $!B$ हे~$w$ येथे
सत्य असल्यामुळेच $\Diamond !B$ हेही~$w$ येथे सत्य असेल
असे नाही. मात्र $Rww$ असेल, उदाहरणार्थ $R$ परावर्ती असेल,
तर ते खरे ठरते. $Rww'$ असलेले $w'$ मुळीच नसेल, तर कोणत्याही
$!A$ साठी $\mSat/{M}{\Diamond !A}[w]$ असते.

\begin{prob}
  \olref[nml][syn][rel]{fig:simple} मधील प्रतिमान पाहा. पुढीलपैकी
  कोणती विधाने खरी आहेत?
    \begin{enumerate}
    \item $\mSat{M}{q}[w_1]$;
    \item $\mSat{M}{\lnot q}[w_3]$;
    \item $\mSat{M}{p \lor q}[w_1]$;
    \item $\mSat{M}{\Box (p \lor q)}[w_1]$;
    \item $\mSat{M}{\Box q}[w_3]$;
    \item $\mSat{M}{\Box \bot}[w_3]$;
    \item $\mSat{M}{\Diamond q}[w_1]$;
    \item $\mSat{M}{\Box q}[w_1]$;
    \item $\mSat{M}{\lnot \Box\Box \lnot q}[w_1]$.
    \end{enumerate}
\end{prob}

% Prove duality of \Box and \Diamond
\begin{tagblock}{prvDiamond}
\begin{prop}\ollabel{prop:dual}
  \begin{enumerate}
  \item $\mSat{M}{\Box !A}[w]$ iff $\mSat{M}{\lnot\Diamond\lnot !A}[w]$.
  \item $\mSat{M}{\Diamond !A}[w]$ iff $\mSat{M}{\lnot\Box\lnot !A}[w]$.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ तेव्हा आणि केवळ
      तेव्हाच, जेव्हा $\mSat/{M}{\Diamond\lnot !A}[w]$;
      हे $\mSat{M}{}[w]$ च्या व्याख्येवरून मिळते.
      $\mSat{M}{\Diamond\lnot !A}[w]$ तेव्हा आणि केवळ तेव्हाच,
      जेव्हा $Rww'$ असलेल्या एखाद्या $w'$ साठी
      $\mSat{M}{\lnot !A}[w']$. म्हणून
      $\mSat/{M}{\Diamond\lnot !A}[w]$ तेव्हा आणि केवळ तेव्हाच,
      जेव्हा $Rww'$ असलेल्या प्रत्येक $w'$ साठी
      $\mSat/{M}{\lnot !A}[w']$. तसेच
      $\mSat/{M}{\lnot !A}[w']$ तेव्हा आणि केवळ तेव्हाच,
      जेव्हा $\mSat{M}{!A}[w']$. मिळून,
      $\mSat{M}{\lnot\Diamond\lnot !A}[w]$ तेव्हा आणि केवळ
      तेव्हाच, जेव्हा $Rww'$ असलेल्या प्रत्येक $w'$ साठी
      $\mSat{M}{!A}[w']$. $\mSat{M}{}[w]$ च्या व्याख्येवरून
      ही अट तेव्हा आणि केवळ तेव्हाच असते, जेव्हा
      $\mSat{M}{\Box !A}[w]$.
    \item \iftag{probDiamond}{सराव.}{$\mSat{M}{\lnot\Box\lnot !A}[w]$
      तेव्हा आणि केवळ तेव्हाच, जेव्हा
      $\mSat/{M}{\Box\lnot !A}[w]$.
      $\mSat{M}{\Box\lnot !A}[w]$ तेव्हा आणि केवळ तेव्हाच,
      जेव्हा $Rww'$ असलेल्या प्रत्येक $w'$ साठी
      $\mSat{M}{\lnot !A}[w']$. म्हणून
      $\mSat/{M}{\Box\lnot !A}[w]$ तेव्हा आणि केवळ तेव्हाच,
      जेव्हा $Rww'$ असलेल्या एखाद्या $w'$ साठी
      $\mSat/{M}{\lnot !A}[w']$. तसेच
      $\mSat/{M}{\lnot !A}[w']$ तेव्हा आणि केवळ तेव्हाच,
      जेव्हा $\mSat{M}{!A}[w']$. मिळून,
      $\mSat{M}{\lnot\Box\lnot !A}[w]$ तेव्हा आणि केवळ
      तेव्हाच, जेव्हा $Rww'$ असलेल्या एखाद्या~$w'$ साठी
      $\mSat{M}{!A}[w']$. पुन्हा $\mSat{M}{}[w]$ च्या
      व्याख्येवरून ही अट तेव्हा आणि केवळ तेव्हाच असते,
      जेव्हा $\mSat{M}{\Diamond !A}[w]$.}
  \end{enumerate}
\end{proof}

\begin{probtag}{probDiamond}
  \olref[nml][syn][trw]{prop:dual} ची सिद्धता पूर्ण करा.
\end{probtag}
\end{tagblock}

\begin{prob}
  $\mModel{M} = \tuple{W, R, V}$ हे प्रतिमान असू द्या आणि
  $w_1, w_2 \in W$ साठी पुढील अटी पूर्ण होतात असे समजा:
  \begin{enumerate}
  \item प्रत्येक !!{propositional variable}~$p$ साठी $w_1 \in V(p)$
    तेव्हा आणि केवळ तेव्हाच, जेव्हा $w_2 \in V(p)$; आणि
  \item प्रत्येक $w \in W$ साठी: $Rw_1w$ तेव्हा आणि केवळ
    तेव्हाच, जेव्हा $Rw_2w$.
  \end{enumerate}
  !!{formula}s वर विगमन करून दाखवा की सर्व !!{formula}s~$!A$
  साठी $\mSat{M}{!A}[w_1]$ तेव्हा आणि केवळ तेव्हाच,
  जेव्हा $\mSat{M}{!A}[w_2]$.
\end{prob}

\begin{probtag}{prvDiamond}
  $\mModel{M} = \tuple{W, R, V}$ असू द्या. दाखवा की
  $\mSat{M}{\lnot\Diamond !A}[w]$ तेव्हा आणि केवळ तेव्हाच,
  जेव्हा $\mSat{M}{\Box\lnot!A}[w]$.
\end{probtag}
