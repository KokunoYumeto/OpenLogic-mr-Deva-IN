\olsection{प्रतिमानात सत्य असणे}

कधीकधी दिलेल्या प्रतिमानातील प्रत्येक जगात कोणती !!{formula}s सत्य
आहेत, हे जाणून घ्यायचे असते. यासाठी एक संकेतन देऊ.

\begin{defn}
  !!^a{formula}~$!A$ हे $M = \tuple{W, R,
    V}$ या \emph{प्रतिमानात सत्य आहे}, म्हणजे $\mSat{M}{!A}$,
  तेव्हा आणि केवळ तेव्हाच, जेव्हा प्रत्येक $w \in W$ साठी
  $\mSat{M}{!A}[w]$.
\end{defn}

\begin{prop}\ollabel{prop:truthfacts}
  \begin{enumerate}
  \item $\mSat{M}{!A}$ असेल, तर $\mSat/{M}{\lnot !A}$;
    पण याचा \emph{व्यत्यास नाही}.
  \item $\mSat{M}{!A \lif !B}$ असेल, तर $\mSat{M}{!A}$
    फक्त $\mSat{M}{!B}$ असेल तरच; पण याचा \emph{व्यत्यास नाही}.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
  \item $\mSat{M}{!A}$ असेल, तर $!A$ हे $W$ मधील सर्व
    जगांत सत्य आहे. $W \neq \emptyset$ असल्यामुळे
    $\mSat{M}{\lnot!A}$ असू शकत नाही; तसे असल्यास
    एखाद्या जगात $!A$ सत्यही आणि असत्यही असावे लागेल.

    दुसरीकडे, $\mSat/{M}{\lnot !A}$ असेल, तर एखाद्या
    $w \in W$ येथे $!A$ सत्य आहे. यावरून \emph{प्रत्येक}
    $w \in W$ साठी $\mSat{M}{!A}[w]$ मिळत नाही. उदाहरणार्थ,
    \olref[rel]{fig:simple} मधील प्रतिमानात $\mSat/{M}{\lnot p}$
    आणि $\mSat/{M}{p}$ ही दोन्ही आहेत.
  \item $\mSat{M}{!A \lif !B}$ आणि $\mSat{M}{!A}$ असे समजा.
    $\mSat{M}{!B}$ दाखवण्यासाठी कोणतेही $w \in W$ घ्या.
    मग $\mSat{M}{!A \lif !B}[w]$ आणि $\mSat{M}{!A}[w]$;
    म्हणून $\mSat{M}{!B}[w]$. $w$ कोणतेही असल्यामुळे
    $\mSat{M}{!B}$.

    व्यत्यास असत्य ठरवण्यासाठी असे प्रतिमान $\mModel{M}$
    शोधावे लागेल की $\mSat{M}{!A}$ हे फक्त $\mSat{M}{!B}$
    असेल तरच, पण $\mSat/{M}{!A \lif !B}$. पुन्हा
    \olref[rel]{fig:simple} मधील प्रतिमान पाहा: $\mSat/{M}{p}$,
    आणि म्हणून $\mSat{M}{p}$ हे फक्त $\mSat{M}{q}$ असेल
    तरच, हे रिक्तपणे खरे आहे. मात्र $w_1$ येथे $p$ सत्य
    आणि $q$ असत्य असल्यामुळे $\mSat/{M}{p \lif
      q}$.
  \end{enumerate}
\end{proof}

\begin{prob}
  $p_1$, $p_2$, $p_3$ हीच !!{propositional variable}s असलेल्या
  भाषेसाठी पुढील प्रतिमान $\mModel{M}$ पाहा:
  \begin{center}
    \begin{tikzpicture}[modal]
      \node[world] (w1) [label={[align=right]left:\mTrue{p_1}\\\mFalse{p_2}\\\mFalse{p_3}}]
            {$w_1$} ;
      \node[world] (w2) [label={[align=right]right:\mTrue{p_1}\\\mTrue{p_2}\\\mFalse{p_3}},
        below right=of w1]  {$w_2$};
      \node[world] (w3) [label={[align=right]right:\mTrue{p_1}\\\mTrue{p_2}\\\mTrue{p_3}},
        above right=of w2] {$w_3$};
      \draw[reflexive above] (w3) to (w3);
      \draw[->] (w1) to (w2);
      \draw[->] (w2) to (w3);
      \draw[->] (w1) to (w3);
    \end{tikzpicture}
  \end{center}
  पुढील !!{formula}s आणि रूपबंध $\mModel{M}$ या प्रतिमानात,
  म्हणजे $\mModel{M}$ मधील प्रत्येक जगात, सत्य आहेत का? स्पष्ट करा.
  \begin{enumerate}
  \item $p\lif \Diamond p$ ($p$ अणुरूप असेल तेव्हा);
  \item $!A\lif \Diamond !A$ ($!A$ कोणतेही असेल तेव्हा);
  \item $\Box p \lif p$ ($p$ अणुरूप असेल तेव्हा);
  \item $\lnot p \lif \Diamond \Box p$ ($p$ अणुरूप असेल तेव्हा);
  \item $\Diamond \Box !A$ ($!A$ कोणतेही असेल तेव्हा);
  \item $\Box \Diamond p$ ($p$ अणुरूप असेल तेव्हा).
  \end{enumerate}
\end{prob}
